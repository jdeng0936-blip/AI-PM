**Pattern (apply to every `select(DailyReport...)` / `.where(...)` chain):**

```python
# Before
stmt = select(DailyReport).where(DailyReport.user_id == user_id)

# After

    DailyReport.user_id == user_id,
    DailyReport.deleted_at.is_(None),  # V2.4 Stage 3 C1
)

    DailyReport.report_date >= start_date,
    DailyReport.report_date <= end_date,
    DailyReport.deleted_at.is_(None),  # V2.4 Stage 3 C1
)
```

@router.patch("/batch-restore")
async def batch_restore_reports(
    body: BatchRestoreBody = Body(...),
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """
    批量恢复已软删日报(SET deleted_at = NULL)。
    权限:admin / manager 任意 ids;employee 仅自己的。
    """
    cond = and_(
        DailyReport.id.in_(body.ids),
        DailyReport.deleted_at.is_not(None),
    )
    if current_user.role == UserRole.employee:
        cond = and_(cond, DailyReport.user_id == current_user.id)

    result = await db.execute(
        update(DailyReport).where(cond).values(deleted_at=None).returning(DailyReport.id)
    )
    restored_ids = [r[0] for r in result.all()]
    await db.commit()

    return {
        "requested": len(body.ids),
        "restored_count": len(restored_ids),
        "restored_ids": [str(i) for i in restored_ids],
    }
```

@router.patch("/batch-restore")
async def batch_restore_projects(
    body: BatchRestoreBody = Body(...),
    db: AsyncSession = Depends(get_db),
    _admin: User = Depends(require_role(UserRole.admin)),  # 仅 admin
):
    """
    批量恢复已软删的临时项目(SET deleted_at = NULL)。
    主干项目的"撤销归档"复用 unarchive_project 端点,不在此处理。

    result = await db.execute(
        update(Project)
        .where(
            Project.id.in_(body.ids),
            Project.deleted_at.is_not(None),
            Project.is_temporary.is_(True),
        )
        .values(deleted_at=None)
        .returning(Project.id)
    )
    restored_ids = [r[0] for r in result.all()]
    await db.commit()

    return {
        "requested": len(body.ids),
        "restored_count": len(restored_ids),
        "restored_ids": [str(i) for i in restored_ids],
    }
```

```typescript
'use client'

import { useEffect, useState, useCallback } from 'react'
import { useRouter } from 'next/navigation'
import { toast } from 'sonner'
import { Trash2 } from 'lucide-react'
import { useAuthStore } from '@/stores/use-auth-store'
import { useMultiSelect } from '@/lib/hooks/use-multi-select'
import { ListActionBar } from '@/components/list-action-bar'
import { getReports, batchRestoreReports } from '@/api/reports'
import { getProjects, batchRestoreProjects } from '@/api/projects'

type Tab = 'reports' | 'projects'

export default function RecycleBinPage() {
  const router = useRouter()
  const { isAdmin } = useAuthStore()
  const [tab, setTab] = useState<Tab>('reports')
  const [reports, setReports] = useState<any[]>([])
  const [projects, setProjects] = useState<any[]>([])
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    if (!isAdmin) {
      toast.error('需 admin 权限')
      router.replace('/')
    }
  }, [isAdmin, router])

  const loadReports = useCallback(async () => {
    setLoading(true)
    try {
      const res: any = await getReports({ include_deleted: true, page_size: 100 })
      setReports(res.items || [])
    } finally {
      setLoading(false)
    }
  }, [])

  const loadProjects = useCallback(async () => {
    setLoading(true)
    try {
      const res: any = await getProjects({ include_deleted: true, page_size: 100 })
      setProjects(Array.isArray(res) ? res : res.items || [])
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => {
    if (!isAdmin) return
    if (tab === 'reports') loadReports()
    else loadProjects()
  }, [tab, isAdmin, loadReports, loadProjects])

  const items = tab === 'reports' ? reports : projects
  const idKey = tab === 'reports' ? 'id' : 'project_id'
  const ms = useMultiSelect(items, { idKey: idKey as any })

  const handleRestore = async () => {
    const ids = Array.from(ms.selectedIds) as string[]
    if (!ids.length) return
    try {
      const fn = tab === 'reports' ? batchRestoreReports : batchRestoreProjects
      const res: any = await fn(ids)
      toast.success(`已恢复 ${res?.restored_count ?? ids.length} 条`)
      ms.clearAll()
      if (tab === 'reports') await loadReports()
      else await loadProjects()
    } catch (e: any) {
      toast.error(e?.response?.data?.detail || '恢复失败')
    }
  }

  if (!isAdmin) return null

  return (
    <div className="p-6 animate-in">
      <h1 className="text-2xl font-semibold mb-4 flex items-center gap-2">
        <Trash2 size={24} /> 回收站
      </h1>
      <p className="text-sm text-slate-400 mb-4">
        显示已软删的日报与临时项目。点击"恢复"可还原。本页仅 admin 可见,不提供永久删除。
      </p>

      {/* Tab 切换 */}
      <div className="flex gap-2 mb-4 border-b" style={{ borderColor: 'var(--color-border-subtle)' }}>
        {(['reports', 'projects'] as Tab[]).map(t => (
          <button
            key={t}
            onClick={() => { setTab(t); ms.clearAll() }}
            className="px-4 py-2 text-sm transition-colors"
            style={{
              color: tab === t ? '#3b82f6' : 'var(--color-text-secondary)',
              borderBottom: tab === t ? '2px solid #3b82f6' : '2px solid transparent',
            }}
          >
            {t === 'reports' ? `日报 (${reports.length})` : `临时项目 (${projects.length})`}
          </button>
        ))}
      </div>

      <ListActionBar selectedCount={ms.selectedCount} onClear={ms.clearAll}>
        <button
          onClick={handleRestore}
          className="px-3 py-1.5 text-sm rounded-md bg-green-600 hover:bg-green-700 text-white"
        >
          🔄 恢复选中
        </button>
      </ListActionBar>

      {/* 列表 */}
      {loading && <div className="text-sm text-slate-400">加载中…</div>}
      {!loading && items.length === 0 && (
        <div className="text-sm text-slate-400 py-8 text-center">回收站为空 ✨</div>
      )}
      {!loading && items.length > 0 && (
        <div className="space-y-2">
          <label className="flex items-center gap-2 text-sm pb-2">
            <input
              type="checkbox"
              checked={ms.isAllSelected}
              ref={el => { if (el) el.indeterminate = ms.isIndeterminate }}
              onChange={e => e.target.checked ? ms.selectAll() : ms.clearAll()}
            />
            全选
          </label>
          {items.map((it: any) => {
            const id = it[idKey]
            return (
              <div
                key={id}
                className="flex items-center gap-3 p-3 rounded-md border"
                style={{ borderColor: 'var(--color-border-subtle)' }}
              >
                <input
                  type="checkbox"
                  checked={ms.isSelected(id)}
                  onChange={() => ms.toggle(id)}
                />
                {tab === 'reports' ? (
                  <div className="flex-1 text-sm">
                    <span className="font-medium">{it.report_date}</span>
                    <span className="mx-2 text-slate-500">·</span>
                    <span>{it.user_name || it.member}</span>
                    <span className="mx-2 text-slate-500">·</span>
                    <span className="text-slate-400 truncate">
                      {(it.parsed_content || it.raw_input_text || '').slice(0, 50)}
                    </span>
                  </div>
                ) : (
                  <div className="flex-1 text-sm">
                    <span className="font-medium">{it.name}</span>
                    <span className="mx-2 text-slate-500">·</span>
                    <span className="text-slate-400">{it.code}</span>
                  </div>
                )}
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}
```
